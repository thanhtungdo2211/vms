##1. Check API Health
curl -s http://localhost:8084/api/health

2. Add Cameras

# Add cam1
curl -s -X POST "http://localhost:8084/api/cameras" \
  -H "Content-Type: application/json" \
  -d '{"camera_id": "cam1", "uri": "rtsp://192.168.6.14:8554/testface1", "branches": ["recognition", "detection"]}'

# Add cam2
curl -s -X POST "http://localhost:8084/api/cameras" \
  -H "Content-Type: application/json" \
  -d '{"camera_id": "cam2", "uri": "rtsp://192.168.6.14:8554/testface2", "branches": ["recognition", "detection"]}'

# Add cam3
curl -s -X POST "http://localhost:8084/api/cameras" \
  -H "Content-Type: application/json" \
  -d '{"camera_id": "cam3", "uri": "rtsp://192.168.6.14:8554/testface3", "branches": ["recognition", "detection"]}'

3. List Cameras

curl -s "http://localhost:8084/api/cameras" | python3 -m json.tool

4. Start Per-Camera RTSP Publishing

# cam1/detection -> rtsp://192.168.6.14:8554/cam1_detection
curl -s -X POST "http://localhost:8084/api/cameras/cam1/branches/detection/rtsp/start" \
  -H "Content-Type: application/json" \
  -d '{"location": "rtsp://192.168.6.14:8554/cam1_detection", "bitrate": 4000000}'

# cam2/recognition -> rtsp://192.168.6.14:8554/cam2_recognition
curl -s -X POST "http://localhost:8084/api/cameras/cam2/branches/recognition/rtsp/start" \
  -H "Content-Type: application/json" \
  -d '{"location": "rtsp://192.168.6.14:8554/cam2_recognition", "bitrate": 4000000}'

# cam3/recognition -> rtsp://192.168.6.14:8554/cam3_recognition
curl -s -X POST "http://localhost:8084/api/cameras/cam3/branches/recognition/rtsp/start" \
  -H "Content-Type: application/json" \
  -d '{"location": "rtsp://192.168.6.14:8554/cam3_recognition", "bitrate": 4000000}'

5. Check Per-Camera RTSP Status

curl -s "http://localhost:8084/api/cameras/rtsp/status" | python3 -m json.tool

6. Stop Per-Camera RTSP

curl -s -X POST "http://localhost:8084/api/cameras/cam1/branches/detection/rtsp/stop"
curl -s -X POST "http://localhost:8084/api/cameras/cam2/branches/recognition/rtsp/stop"

7. View Streams (ffplay)

ffplay rtsp://192.168.6.14:8554/cam1_detection
ffplay rtsp://192.168.6.14:8554/cam2_recognition
